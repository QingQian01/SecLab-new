```kotlin
fun portMap(name: String): List<String> {
    return try {
        val inspect = dockerClient.inspectContainerCmd(name).exec()
        val ports = inspect.networkSettings.ports
        if (ports == null || ports.bindings.isEmpty()) {
            return emptyList()
        }
        ports.bindings.entries.flatMap { (exposed, bindings) ->
            val exposedStr = exposed.toString()
            if (bindings.isNullOrEmpty()) {
                listOf("$exposedStr (no binding)")
            } else {
                bindings.map { binding ->
                    val host = binding.hostIp ?: "0.0.0.0"
                    val port = binding.hostPort ?: ""
                    "$exposedStr -> $host:$port"
                }
            }
        }
    } catch (e: NotFoundException) {
        throw DockerSdkException("container '$name' not found", stderr = e.message ?: "")
    } catch (e: Exception) {
        throw DockerSdkException("port map failed for '$name': ${e.message}", cause = e)
    }
}
```

**Why**：`Ports.getBindings()` 返回 `Map<ExposedPort, PortBinding[]>`，绕过 `Ports` 不是 `Map` 子类的类型问题；同时处理 null 与空 binding 两种边界。

```python
def ensure_guidance_conversation(
    *,
    lab_session_id: str,
    context: dict[str, Any],
    source: str = "guidance-proactive",
) -> dict[str, Any]:
    """
    为 proactive guidance 信号查找或创建本地会话。

    - 按 `lab_session_id + source='guidance-proactive'` 查找；找到则 UPDATE last_message_at。

    ensure_ai_schema()
    now = datetime.utcnow()
    fixed_id = f"guidance-{lab_session_id}"

    with get_connection() as conn:
        with conn.cursor() as cursor:
            cursor.execute(
                """
                SELECT conversation_id, dify_conversation_id
                FROM ai_conversation
                WHERE lab_session_id = %(lab_session_id)s AND source = %(source)s
                LIMIT 1
                """,

            )
            row = cursor.fetchone()
            if row:
                cursor.execute(
                    """
                    UPDATE ai_conversation
                    SET user_id = COALESCE(%(user_id)s, user_id),
                        class_id = COALESCE(%(class_id)s, class_id),

                        module_id = COALESCE(%(module_id)s, module_id),
                        task_id = COALESCE(%(task_id)s, task_id),
                        last_message_at = %(now)s,
                        status = 'active'
                    WHERE conversation_id = %(conversation_id)s
                    """,

                )
                conn.commit()
                return {
                    "conversation_id": row["conversation_id"],
                    "dify_conversation_id": row.get("dify_conversation_id"),
                }

            cursor.execute(
```

#### 2.2.1 新建 `d:\SecLab\SecLab\seclab-backend\ai-agent-service\guidance_stream_registry.py`

```python
"""SSE 订阅广播注册表（单 worker 部署约束）。"""
import asyncio
import logging
from typing import Any

logger = logging.getLogger(__name__)

# lab_session_id -> set of subscriber queues
_queues: dict[str, set[asyncio.Queue]] = {}
_lock = asyncio.Lock()


async def register(lab_session_id: str) -> asyncio.Queue:
    """订阅指定 lab_session_id 的 guidance 推送；返回 asyncio.Queue。"""
    async with _lock:
        queue: asyncio.Queue = asyncio.Queue(maxsize=64)
        _queues.setdefault(lab_session_id, set()).add(queue)
        logger.info("guidance stream registered lab_session_id=%s total=%d",
                    lab_session_id, len(_queues[lab_session_id]))
        return queue


async def unregister(lab_session_id: str, queue: asyncio.Queue) -> None:
    """取消订阅；若该 lab_session_id 已无订阅者，清理 key。"""
    async with _lock:
        subs = _queues.get(lab_session_id)
        if not subs:
            return
        subs.discard(queue)
        if not subs:
            _queues.pop(lab_session_id, None)
        logger.info("guidance stream unregistered lab_session_id=%s remaining=%d",
                    lab_session_id, len(subs))


async def broadcast(lab_session_id: str, event: dict[str, Any]) -> None:
    """向该 lab_session_id 的所有订阅者推送 event（队列满则丢弃，避免反压）。"""
    async with _lock:
        subs = list(_queues.get(lab_session_id, set()))
    for queue in subs:
        try:
            queue.put_nowait(event)
        except asyncio.QueueFull:
            logger.warning("guidance queue full lab_session_id=%s, dropping event", lab_session_id)


async def active_subscriber_count(lab_session_id: str) -> int:
    async with _lock:
        return len(_queues.get(lab_session_id, set()))
```

```python
from guidance_service import GuidanceSignal, handle_guidance_signal
from guidance_stream_registry import (
    register as register_guidance_stream,
    unregister as unregister_guidance_stream,
)
import asyncio


@app.post("/api/guidance/signal")
async def guidance_signal_endpoint(
    signal: GuidanceSignal,
    _: None = Security(verify_tool_key),
):
    """
    接收 docker-service 推送的挫败信号，流式调用 Dify 生成启发式提示。
    docker-service 的 GuidanceSignalClient 用 RestTemplate POST（fire-and-forget），
    本端点返回 SSE 流；docker-service 仅记录 status code，不消费 body。
    """
    return StreamingResponse(
        handle_guidance_signal(signal),
        media_type="text/event-stream",
    )


@app.get("/api/guidance/stream/{lab_session_id}")
async def guidance_stream_endpoint(lab_session_id: str):
    """
    前端订阅此 SSE 端点，接收该 lab_session_id 的所有 guidance_alert 推送。
    断开自动注销订阅。
    """
    queue = await register_guidance_stream(lab_session_id)

    async def event_generator():
        try:
            yield f"event: connected\ndata: {json.dumps({'lab_session_id': lab_session_id, 'ts': datetime.utcnow().isoformat() + 'Z'})}\n\n"
            while True:
                payload = await queue.get()
                yield f"event: guidance_alert\ndata: {json.dumps(payload, ensure_ascii=False, default=str)}\n\n"
        except asyncio.CancelledError:
            pass
        finally:
            await unregister_guidance_stream(lab_session_id, queue)

    return StreamingResponse(event_generator(), media_type="text/event-stream")
```

**Why**：
- `POST /api/guidance/signal`：鉴权（`verify_tool_key`）+ SSE 流式返回 Dify 启发式提示。
- `GET /api/guidance/stream/{lab_session_id}`：前端订阅广播通道，断开自动注销。

## 剩余风险与需用户确认事项

1. **Dify Agent 是否已配置 guidance 应用**：本计划假设 Dify 云端已有对应 Agent 与 RAG 知识库；若未配置，`/api/guidance/signal` 会返回 Dify 404，需用户在 Dify 后台配置。
2. **Docker daemon 可达性**：Windows 环境默认 `npipe:////./pipe/docker_engine`；若 Docker Desktop 未运行，`DockerSdkConfig.dockerClient()` 初始化会失败。需确认 Docker Desktop 已启动。
3. **`AGENT_SIGNAL_TOKEN` 一致性**：docker-service 的 `seclab.agent.signal-token` 与 ai-agent-service 的 `TOOL_API_KEY` 必须一致，否则信号被 403 拒绝。建议两边都留空（不鉴权）或都设为同一值。
4. **多 worker 部署**：当前 `guidance_stream_registry` 是进程内广播；若 ai-agent-service 用 `uvicorn --workers 4`，跨 worker 信号无法广播。生产部署需替换为 Redis pubsub（评审缺口 14）。
5. **测试中的 mock 范围**：`test_guidance_service.py` 需 mock `httpx.AsyncClient.stream`、`save_error_event`、`ensure_guidance_conversation`、`save_ai_message`、`save_context_injection`、`guidance_stream_registry.broadcast`；若 mock 不全可能误报失败。
