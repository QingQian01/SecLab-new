#### 1.3.2 新建 `TtyLogHandler.kt`

路径：`d:\SecLab\SecLab\seclab-backend\docker-service\src\main\kotlin\me\myot233\seclab\dockerservice\TtyLogHandler.kt`

`TextWebSocketHandler` 实现：
- `afterConnectionEstablished`：从 URI 提取 `containerName`，校验 `LabAllowlist.isAllowed`；从 query string 提取 `lab_session_id` / `user_id`（评审缺口 4 修复）；`sessionKey = labSessionId ?: "${containerName}:${userId ?: 'anon'}"`；创建 `Subscriber`（带 `onTrigger` 回调）订阅 `TtyLogStreamService`。
- `handleTextMessage`：仅支持 `{"type":"ping"}` 心跳，回 `{"type":"pong","ts":"..."}`；其余客户端→服务端文本忽略。
- `afterConnectionClosed`：try-catch 兜底 `unsubscribe`（评审缺口 6 修复）。
- `handleTrigger(decision, context)`：把 `TriggerDecision` 转 `GuidanceSignal`，调 `guidanceSignalClient.sendSignal { ... }` 异步发送。
- 推送帧格式：`{"ts":"...","stream":"stdout|stderr","line":"..."}`。

```kotlin
package me.myot233.seclab.dockerservice

import com.fasterxml.jackson.module.kotlin.jacksonObjectMapper
import org.slf4j.LoggerFactory
import org.springframework.stereotype.Component
import org.springframework.web.socket.CloseStatus
import org.springframework.web.socket.TextMessage
import org.springframework.web.socket.WebSocketSession
import org.springframework.web.socket.handler.TextWebSocketHandler
import org.springframework.web.util.UriComponentsBuilder
import java.util.concurrent.ConcurrentHashMap

@Component
class TtyLogHandler(
    private val ttyLogStreamService: TtyLogStreamService,
    private val guidanceSignalClient: GuidanceSignalClient,
) : TextWebSocketHandler() {

    private val log = LoggerFactory.getLogger(javaClass)
    private val objectMapper = jacksonObjectMapper().findAndRegisterModules()
    private val sessionContexts = ConcurrentHashMap<WebSocketSession, TtySessionContext>()

    override fun afterConnectionEstablished(session: WebSocketSession) {
        val containerName = extractContainerName(session) ?: run {
            session.close(CloseStatus.POLICY_VIOLATION); return
        }
        if (!LabAllowlist.isAllowed(containerName)) {
            session.close(CloseStatus.POLICY_VIOLATION); return
        }
        val params = extractQueryParams(session)
        val sessionKey = params.labSessionId ?: "${containerName}:${params.userId ?: "anon"}"
        val context = TtySessionContext(containerName, sessionKey, params.labSessionId, params.userId)
        sessionContexts[session] = context

        val subscriber = TtyLogStreamService.Subscriber(
            sink = TtyLogStreamService.TtyLogSink { frame -> broadcastFrame(session, frame) },
            sessionKey = sessionKey,
            onTrigger = { decision -> handleTrigger(decision, context) },
        )
        context.subscriber = subscriber
        ttyLogStreamService.subscribe(containerName, subscriber)
    }

    override fun handleTextMessage(session: WebSocketSession, message: TextMessage) {
        val payload = message.payload.trim()
        if (payload.contains("\"ping\"")) {
            try { session.sendMessage(TextMessage("""{"type":"pong"}""")) } catch (_: Exception) {}
        }
    }

    override fun afterConnectionClosed(session: WebSocketSession, status: CloseStatus) {
        val context = sessionContexts.remove(session) ?: return
        try {
            context.subscriber?.let { ttyLogStreamService.unsubscribe(context.containerName, it) }
        } catch (e: Exception) {
            log.warn("unsubscribe failed: {}", e.message)
        }
    }

    private fun broadcastFrame(session: WebSocketSession, frame: TtyLogStreamService.TtyLogFrame) {
        try {
            session.sendMessage(TextMessage(objectMapper.writeValueAsString(frame)))
        } catch (e: Exception) {
            try { session.close(CloseStatus.SERVER_ERROR) } catch (_: Exception) {}
        }
    }

    internal fun handleTrigger(decision: FrustrationTracker.TriggerDecision, context: TtySessionContext) {
        val target = LabAllowlist.targetOf(context.containerName)
        val signalId = guidanceSignalClient.newSignalId()
        guidanceSignalClient.sendSignal {
            GuidanceSignal(
                signal_id = signalId,
                request_id = "tty-$signalId",
                trigger_reason = decision.reason,
                lab_session_id = context.labSessionId,
                user_id = context.userId,
                class_id = null, course_id = null, module_id = null, task_id = null,
                container_name = context.containerName,
                lab_key = LabAllowlist.labKeyOf(context.containerName) ?: "unknown",
                error_slice = decision.recentErrors.lastOrNull() ?: "",
                exit_code = null,
                recent_commands = decision.recentCommands,
                lab_task_summary = target?.labTaskSummary ?: "",
                frustration_score = decision.score,
                locale = "zh-CN",
                triggered_at = decision.triggeredAt,
            )
        }
    }

    private fun extractContainerName(session: WebSocketSession): String? {
        val path = session.uri?.path?.trimEnd('/') ?: return null
        val idx = path.lastIndexOf('/')
        return if (idx >= 0 && idx < path.length - 1) path.substring(idx + 1) else null
    }

    private fun extractQueryParams(session: WebSocketSession): QueryParams {
        val params = UriComponentsBuilder.fromUri(session.uri).build().queryParams
        return QueryParams(
            labSessionId = params.getFirst("lab_session_id")?.takeIf { it.isNotBlank() },
            userId = params.getFirst("user_id")?.takeIf { it.isNotBlank() }?.toLongOrNull(),
        )
    }

    private data class QueryParams(val labSessionId: String? = null, val userId: Long? = null)
    internal data class TtySessionContext(
        val containerName: String, val sessionKey: String,
        val labSessionId: String?, val userId: Long?,
        var subscriber: TtyLogStreamService.Subscriber? = null,
    )
}
```

**Why**：
- 握手校验白名单 + 提取 query 参数（评审缺口 4 修复）。
- `afterConnectionClosed` 用 try-catch 兜底 unsubscribe（评审缺口 6 修复）。
- 通过 `handleTrigger` 把 `TriggerDecision` 转 `GuidanceSignal` 异步发送，不阻塞 WS 推送。
- `handleTextMessage` 支持 ping/pong 心跳。

```python
from guidance_service import GuidanceSignal, handle_guidance_signal
from guidance_stream_registry import register as register_guidance_stream, unregister as unregister_guidance_stream

@app.post("/api/guidance/signal")
async def guidance_signal_endpoint(
    signal: GuidanceSignal,
    _: None = Security(verify_tool_key),
):
    """接收 docker-service 推送的挫败信号，流式调用 Dify 生成启发式提示。"""
    return StreamingResponse(handle_guidance_signal(signal), media_type="text/event-stream")

@app.get("/api/guidance/stream/{lab_session_id}")
async def guidance_stream_endpoint(lab_session_id: str):
    """前端订阅此 SSE 端点，接收该 lab_session_id 的所有 guidance_alert 推送。"""
    queue = await register_guidance_stream(lab_session_id)

    async def event_generator():
        try:
            yield f"event: connected\ndata: {json.dumps({'lab_session_id': lab_session_id, 'ts': datetime.utcnow().isoformat()})}\n\n"
            while True:
                payload = await queue.get()
                yield f"event: guidance_alert\ndata: {payload}\n\n"
        except asyncio.CancelledError:
            pass
        finally:
            await unregister_guidance_stream(lab_session_id, queue)

    return StreamingResponse(event_generator(), media_type="text/event-stream")
```

**Why**：
- `POST /api/guidance/signal`：鉴权（`verify_tool_key`）+ SSE 流式返回 Dify 启发式提示。
- `GET /api/guidance/stream/{lab_session_id}`：前端订阅广播通道，断开自动注销。

## Assumptions & Decisions（假设与决策）

1. **复用现有 5 张表**：`error_event`、`container_command_event`、`ai_conversation`、`ai_message`、`ai_context_injection`。仅扩展枚举值，不新增表。
2. **Python 侧 `classify_error_excerpt` 是权威错误分类源**：Kotlin 侧 `FrustrationTracker.extractSignature` 仅做轻量判定用于评分，不传给 Dify。
3. **`conversation_id = "guidance-{lab_session_id}"`** 是确定性的，便于跨信号复用；不侵入 `dify_conversation_id` UNIQUE 约束。
4. **单 worker 部署约束**：`guidance_stream_registry` 用进程内 asyncio.Queue；多 worker 部署需替换为 Redis pubsub（代码留 TODO）。
5. **WS TTY 端点不校验 user_id 真实性**：仅校验 containerName ∈ LabAllowlist；公网部署需补 token 校验。
6. **挫败指数阈值初始值**：`threshold=3`、`cooldown=90s`、`repeat-window=300s`，需基于真实数据调参。
7. **Dify RAG 知识库内容**：用户已在云端配置，本方案只负责把 `error_slice`/`lab_key` 等通过 `inputs` 字段传入。
8. **lab_task_summary 当前为静态**：写在 `LabAllowlist` 里；后续可改为从 `experiment-module-service` 动态拉取。
9. **`POST /api/docker/frustration/{lab}` 是降级路径**：文档标注"不要与 WS 同时启用，避免双源触发"。
10. **ephemeral 消息不持久化**：AI 主动提示不写入 localStorage，浏览器刷新后不残留。

## 剩余风险与需用户确认事项

1. **多 worker 部署**：当前 `main.py` 用 `uvicorn.reload=True` 单 worker。若生产用 `gunicorn -w 4`，`guidance_stream_registry` 的 in-process Queue 会跨 worker 失效。**默认按单 worker 部署约束**。
2. **Dify RAG 知识库内容**：用户已在云端配置，本方案不负责内容维护。
3. **lab_task_summary 内容来源**：当前从 `lab_key` 反查静态摘要。若需动态从 `experiment-module-service` 拉取，需额外接口调用。
4. **挫败指数阈值调参**：`threshold=3` 等是初始值，需基于真实学生数据调参。
5. **WS TTY 鉴权**：当前 WS 端点仅校验 `containerName ∈ LabAllowlist`，不校验 `user_id` 真实性。本地/内网部署可接受。
