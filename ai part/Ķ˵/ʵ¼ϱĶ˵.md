### 3.1 `src/composables/useLabState.ts`

**改动前：** 仅保存容器名和启动时间，无事件追踪能力。

**改动后：** 完整的实验状态管理，支持事件收集。

```typescript
import { ref } from 'vue'

export const CONTAINER_NAMES: Record<number, string> = {
  1: 'sqli-lab-web-1',
  2: 'xss-lab-web-1',
  3: 'csrf-lab-web-1',
  4: 'cmd-inject-web-1',
  5: 'upload-lab-web-1',
  6: 'dir-lab-web-1',
}

export const currentLabContainer = ref<string>('')
export const labStartTime = ref<Date | null>(null)
export const isLabRunning = ref<boolean>(false)  // ← 新增

// ↓ 新增：事件接口
export interface LabEvent {
  event_id: string
  event_type: string
  event_time: string
  [key: string]: any
}

const labEvents = ref<LabEvent[]>([])  // ← 新增
let labSessionId = ''                   // ← 新增

function generateId(prefix: string) {   // ← 新增
  return `${prefix}-${Date.now()}-${Math.random().toString(36).substring(2, 8)}`
}

export function setLabRunning(containerName: string) {
  currentLabContainer.value = containerName
  labStartTime.value = new Date()
  isLabRunning.value = true             // ← 新增
  labSessionId = `lab-${Date.now()}`    // ← 新增
  labEvents.value = []                  // ← 新增
  labEvents.value.push({                // ← 新增：自动记录 LAB_START
    event_id: generateId('evt-lab-start'),
    event_type: 'LAB_START',
    event_time: new Date().toISOString(),
    container_name: containerName,
    extra: {}
  })
}

export function clearLab() {
  currentLabContainer.value = ''
  labStartTime.value = null
  isLabRunning.value = false            // ← 新增
  labSessionId = ''                     // ← 新增
  labEvents.value = []                  // ← 新增
}

// ↓ 以下全部为新增导出函数
export function getLabSessionId() {
  return labSessionId
}

export function addLabEvent(event: Partial<LabEvent> & { event_type: string }) {
  labEvents.value.push({
    event_id: event.event_id || generateId('evt'),
    event_type: event.event_type,
    event_time: new Date().toISOString(),
    ...event
  })
}

export function getLabEvents() {
  return labEvents.value
}

export function getLabDurationSeconds() {
  if (!labStartTime.value) return 0
  return Math.round((Date.now() - labStartTime.value.getTime()) / 1000)
}
```

**新增导出说明：**

| 导出 | 类型 | 用途 |
|------|------|------|
| `isLabRunning` | `ref<boolean>` | 全局标记实验是否进行中，路由守卫使用 |
| `LabEvent` | 接口 | 事件数据结构定义 |
| `addLabEvent()` | 函数 | 组件内记录事件（FLAG_SUBMIT 等） |
| `getLabEvents()` | 函数 | 获取当前实验所有事件，用于构建上报 payload |
| `getLabSessionId()` | 函数 | 获取当前实验会话 ID |
| `getLabDurationSeconds()` | 函数 | 获取实验持续秒数 |

```typescript
export interface EventsIngestPayload {
    batch_id: string

        class_id: number

    }
    events: Array<{
        event_id: string
        request_id?: string
        event_type: string
        event_time: string
        user_id: number
        class_id: number
        course_id: number
        module_id: number
        task_id?: number | null
        question_id?: number | null
        lab_session_id: string
        source: string
        question_type?: string | null
        answer?: string | string[] | null
        standard_answer?: string | null
        question_score?: number | null
        cost_time?: number | null
        is_correct?: boolean | null
        container_name?: string | null
        flag_text?: string | null
        score?: number | null
        command_id?: string | null
        command?: string | null
        cwd?: string | null
        exit_code?: number | null
        duration_ms?: number | null
        output_digest?: string | null
        file_path?: string | null
        file_ext?: string | null
        action?: string | null
        size_before?: number | null
        size_after?: number | null
        is_key_file?: boolean | null
        error_signature?: string | null
        error_category?: string | null
        raw_excerpt?: string | null
        severity?: string | null
        extra?: Record<string, any>
    }>
    ai_generated_questions?: Array<unknown>
    profile_rebuild?: {
        user_id: number
        class_id: number
        rebuild_after_ingest: boolean
    }
}

export async function ingestEvents(payload: EventsIngestPayload): Promise<Result<unknown>> {
    const AGENT_BASE_URL = import.meta.env.VITE_AGENT_BASE_URL || "http://localhost:8010";
    try {
        const response = await axios.post(`${AGENT_BASE_URL}/api/external/events/ingest`, payload, {
            headers: { 'Content-Type': 'application/json' }
        })
        console.log('[EventsIngest] 成功:', response.data)
        return response.data
    } catch (error) {
        console.error('[EventsIngest] 失败:', error)
        return { isSuccess: 0, status: 500, message: '事件上报失败', data: null }
    }
}
```

**说明：**
- `EventsIngestPayload` 完整对应 `events_ingest_no_question_submit (1).json` 的 JSON 结构
- `ingestEvents()` 发送 POST 请求到 `VITE_AGENT_BASE_URL/api/external/events/ingest`，请求头 `Content-Type: application/json`
- `VITE_AGENT_BASE_URL` 默认 `http://localhost:8010`（Python FastAPI Agent 服务），最终请求地址为 `http://localhost:8010/api/external/events/ingest`

## 四、三层导航保护机制

| 层级 | 守卫方式 | 触发场景 | 表现 |
|------|----------|----------|------|
| 1 | `onBeforeRouteLeave` | 组件内路由跳转 | 自定义美观弹窗（DaisyUI modal） |

| 3 | `beforeunload` | 关闭/刷新浏览器 | 浏览器原生离开提示 |

### 规范改进要点（对比原始版本）

| 改进项 | 原始版本 | 规范化后 |
|--------|----------|----------|
| 响应 schema | 全部 `type: object` 无具体字段 | 完整定义每个字段（类型、描述、示例、required） |
| 错误响应 | 缺失 | 每个操作补充 400/404/409/500 + `ErrorResponse` schema |
| 可复用组件 | 无 `components/schemas` | 7 个 schema + 2 个 parameter 定义 |
| 请求体验证 | `ExecRequest` 缺 `required` | 所有 `required` 字段明确标注 |
| 参数约束 | `container_name` 无格式校验 | 添加 `pattern: '^[a-zA-Z0-9][a-zA-Z0-9_.-]+$'` |
| 安全方案 | 仅 `ApiKeyAuth` | 新增 `BearerAuth`（Dify 默认方式） |
| 标签分组 | 无 | 3 个 tag：Container Query / Control / Execution |
| 示例值 | 无 | 每个字段添加 `examples` |
| 事件类型 | 无枚举 | `event_type` 定义 11 种枚举值 |
| 文件结构 | 单文件内混合注释 | 独立规范文件，YAML 格式，可被 Swagger Editor 直接校验 |

## 八、注意事项

1. **API 地址**：`ingestEvents` 使用 `VITE_AGENT_BASE_URL`（默认 `http://localhost:8010`），最终请求为 `http://localhost:8010/api/external/events/ingest`。该端点在 Python FastAPI Agent 服务上，而非 Spring Boot 后端（8083）。

2. **用户信息获取**：当前从 `localStorage` 的 `seclab_accounts` 中匹配 token 获取 `userId` 和 `classId`。如果后续有更规范的用户信息存储方式（如 Pinia store），需同步修改 `getUserContext()`。

3. **容器停止**：前端调用 Docker API（Node.js，端口 3000）的 `/api/docker/stop/{container_name}` 停止容器，该接口已在 `docker-api-server.js` 中补全。

4. **Agent 工具部署**：Python 工具代码需在 Dify 平台作为自定义工具部署，`DOCKER_API_BASE` 和 `EVENTS_INGEST_URL` 需根据实际部署地址修改。

5. **后端 Docker SDK 数据补充**：`/api/external/events/ingest` 端点在接收前端事件后，会自动通过 Docker SDK 采集容器运行时数据（CPU/内存/网络/磁盘、进程列表、日志解析事件），与前端事件去重合并后存储。详见 [后端Docker-SDK数据采集与事件上报.md](后端Docker-SDK数据采集与事件上报.md)。
